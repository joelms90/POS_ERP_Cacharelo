# Phase 1 Data Model: Purchase Orders & Inventory Requisitions

This document describes the database tables for PostgreSQL (Drizzle ORM) and the local Isar database collections for the Flutter POS client.

---

## 1. Cloud Database Schema (PostgreSQL via Drizzle)

### 1.1 Purchase Orders Table (`purchase_orders`)

Represents a formal stock request sent to a supplier.

```typescript
export const purchaseOrders = pgTable('purchase_orders', {
  id: uuid('id').primaryKey().defaultRandom(),
  businessId: uuid('business_id').references(() => businesses.id).notNull(),
  providerId: uuid('provider_id').references(() => providers.id).notNull(),
  warehouseId: uuid('warehouse_id').references(() => warehouses.id), // Destination warehouse
  branchId: uuid('branch_id').references(() => branches.id), // Destination branch (if directly to branch)
  poNumber: varchar('po_number', { length: 50 }).unique().notNull(),
  status: varchar('status', { length: 20 }).default('draft').notNull(), // 'draft', 'sent', 'partially_received', 'received', 'cancelled'
  orderDate: timestamp('order_date', { withTimezone: true }).defaultNow().notNull(),
  expectedDeliveryDate: timestamp('expected_delivery_date', { withTimezone: true }),
  actualDeliveryDate: timestamp('actual_delivery_date', { withTimezone: true }),
  subtotal: decimal('subtotal', { precision: 10, scale: 2 }).default('0.00'),
  tax: decimal('tax', { precision: 10, scale: 2 }).default('0.00'),
  total: decimal('total', { precision: 10, scale: 2 }).default('0.00'),
  notes: text('notes'),
  createdBy: uuid('created_by').references(() => users.id).notNull(),
  createdAt: timestamp('created_at', { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp('updated_at', { withTimezone: true }).defaultNow().notNull(),
  deletedAt: timestamp('deleted_at', { withTimezone: true }),
  version: integer('version').default(1).notNull(),
});
```

### 1.2 Purchase Order Items Table (`purchase_order_items`)

Represents line items in a Purchase Order.

```typescript
export const purchaseOrderItems = pgTable('purchase_order_items', {
  id: uuid('id').primaryKey().defaultRandom(),
  poId: uuid('po_id').references(() => purchaseOrders.id).notNull(),
  productId: uuid('product_id').references(() => products.id).notNull(),
  quantityOrdered: decimal('quantity_ordered', { precision: 12, scale: 3 }).notNull(),
  quantityReceived: decimal('quantity_received', { precision: 12, scale: 3 }).default('0.000').notNull(),
  unitCostEstimated: decimal('unit_cost_estimated', { precision: 10, scale: 2 }).notNull(),
  unitCostActual: decimal('unit_cost_actual', { precision: 10, scale: 2 }).default('0.00').notNull(),
});
```

### 1.3 Branch Requisitions Table (`branch_requisitions`)

Represents branch stock requests to a central warehouse.

```typescript
export const branchRequisitions = pgTable('branch_requisitions', {
  id: uuid('id').primaryKey().defaultRandom(),
  businessId: uuid('business_id').references(() => businesses.id).notNull(),
  fromBranchId: uuid('from_branch_id').references(() => branches.id).notNull(),
  toWarehouseId: uuid('to_warehouse_id').references(() => warehouses.id).notNull(),
  status: varchar('status', { length: 20 }).default('pending_approval').notNull(), // 'pending_approval', 'approved', 'standby', 'rejected', 'completed'
  notes: text('notes'),
  createdBy: uuid('created_by').references(() => users.id).notNull(),
  createdAt: timestamp('created_at', { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp('updated_at', { withTimezone: true }).defaultNow().notNull(),
  version: integer('version').default(1).notNull(),
});
```

### 1.4 Branch Requisition Items Table (`branch_requisition_items`)

Represents individual items in a branch requisition.

```typescript
export const branchRequisitionItems = pgTable('branch_requisition_items', {
  id: uuid('id').primaryKey().defaultRandom(),
  requisitionId: uuid('requisition_id').references(() => branchRequisitions.id).notNull(),
  productId: uuid('product_id').references(() => products.id).notNull(),
  quantityRequested: decimal('quantity_requested', { precision: 12, scale: 3 }).notNull(),
  quantityApproved: decimal('quantity_approved', { precision: 12, scale: 3 }).default('0.000').notNull(),
});
```

## 2. Local Cache Schema (JSON/Shared Preferences via SecureStorageService)

Used to enable offline creation and caching of requisitions on the Flutter POS client.

```dart
class LocalBranchRequisition {
  final String id; // UUID v4 generated by the client
  final String fromBranchId;
  final String toWarehouseId;
  final String status; // 'pending_approval'
  final String? notes;
  final DateTime createdAt;
  final String syncState; // 'pending', 'synced', 'failed'
  final List<LocalBranchRequisitionItem> items;

  LocalBranchRequisition({
    required this.id,
    required this.fromBranchId,
    required this.toWarehouseId,
    required this.status,
    this.notes,
    required this.createdAt,
    required this.syncState,
    required this.items,
  });

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'fromBranchId': fromBranchId,
      'toWarehouseId': toWarehouseId,
      'status': status,
      'notes': notes,
      'createdAt': createdAt.toIso8601String(),
      'syncState': syncState,
      'items': items.map((i) => i.toMap()).toList(),
    };
  }

  factory LocalBranchRequisition.fromMap(Map<String, dynamic> map) {
    return LocalBranchRequisition(
      id: map['id'],
      fromBranchId: map['fromBranchId'],
      toWarehouseId: map['toWarehouseId'],
      status: map['status'],
      notes: map['notes'],
      createdAt: DateTime.parse(map['createdAt']),
      syncState: map['syncState'],
      items: (map['items'] as List)
          .map((i) => LocalBranchRequisitionItem.fromMap(i))
          .toList(),
    );
  }
}

class LocalBranchRequisitionItem {
  final String id;
  final String requisitionId;
  final String productId;
  final double quantityRequested;

  LocalBranchRequisitionItem({
    required this.id,
    required this.requisitionId,
    required this.productId,
    required this.quantityRequested,
  });

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'requisitionId': requisitionId,
      'productId': productId,
      'quantityRequested': quantityRequested,
    };
  }

  factory LocalBranchRequisitionItem.fromMap(Map<String, dynamic> map) {
    return LocalBranchRequisitionItem(
      id: map['id'],
      requisitionId: map['requisitionId'],
      productId: map['productId'],
      quantityRequested: (map['quantityRequested'] as num).toDouble(),
    );
  }
}
```
